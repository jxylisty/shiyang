# 端点规模
cd backend && python -X utf8 -c "import sys; sys.path.insert(0,'app'); import main; \
  s=main.app.openapi(); print('paths', len(s['paths']), 'endpoints', sum(len(v) for v in s['paths'].values()))"

# 测试基线
cd backend && python -X utf8 -m pytest tests/ -q

# 知识库行数（只读）
python -X utf8 -c "
import sqlite3
con=sqlite3.connect('file:data/shiyang.db?mode=ro',uri=True)
for t in ('herb','compound','target','rel_compound_target','recipe','compound_map'):
    print(t, con.execute(f'select count(*) from {t}').fetchone()[0])
print('prob union', con.execute('select count(*) from compound where coalesce(prob_cctcm,prob_cctcm_v2,prob_herb_v2) is not null').fetchone()[0])
"

# 文档数
ls docs/*.md | wc -l
